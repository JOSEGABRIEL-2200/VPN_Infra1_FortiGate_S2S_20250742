! Simula el proveedor de Internet: entrega IPs publicas a los FortiGate
! y les da salida a Internet con NAT (PAT) por Ethernet0/2.
! No conoce las redes privadas 10.7.42.x (igual que el Internet real).
! Claves de laboratorio (cambiarlas en un entorno real).
! =====================================================================
enable
configure terminal
hostname ISP
no ip domain-lookup
!
! --- Direccionamiento y NAT ---
interface Ethernet0/0
 description Fortinet1-port2 (IP publica sitio Usuarios)
 ip address 200.7.42.1 255.255.255.252
 ip nat inside
 no shutdown
interface Ethernet0/1
 description Fortinet2-port2 (IP publica sitio Servidor)
 ip address 200.7.42.5 255.255.255.252
 ip nat inside
 no shutdown
interface Ethernet0/2
 description INTERNET (Cloud0 - NAT de VMware)
 ip address dhcp
 ip nat outside
 no shutdown
interface Ethernet0/3
 description SIN-USO
 shutdown
exit
access-list 1 permit 200.7.42.0 0.0.0.7
ip nat inside source list 1 interface Ethernet0/2 overload
!
! --- Configuracion basica de seguridad ---
service password-encryption
enable secret cisco123
banner motd #
 ISP - Acceso restringido. Solo personal autorizado.
 Infraestructura 1 - VPN Site-to-Site - 2025-0742
#
ip domain-name itla.lab
username admin privilege 15 secret cisco123
crypto key generate rsa general-keys modulus 2048
ip ssh version 2
line console 0
 password cisco
 login
 logging synchronous
 exec-timeout 10 0
line vty 0 4
 login local
 transport input ssh
 exec-timeout 10 0
!
end
